<head>
   <title>Store and Retrieve Image</title>
</head>
<body>
   <img id="myImage" src="path/to/your/image.png" alt="Image to store">
   <button onclick="storeImage()">Store Image</button>

   <img id="myStoredImage" alt="Stored Image">
   <button onclick="retrieveImage()">Retrieve Image</button>

   <script>
       function getBase64Image(img) {
           const canvas = document.createElement("canvas");
           canvas.width = img.width;
           canvas.height = img.height;
           const ctx = canvas.getContext("2d");
           ctx.drawImage(img, 0, 0);
           const dataURL = canvas.toDataURL("image/png");
           return dataURL.replace(/^data:image\/(png|jpg);base64,/, "");
       }

       function storeImage() {
           const img = document.getElementById("myImage");
           const base64Image = getBase64Image(img);
           localStorage.setItem("storedImage", base64Image);
       }

       function retrieveImage() {
           const base64Image = localStorage.getItem("storedImage");
           const img = document.getElementById("myStoredImage");
           img.src = "data:image/png;base64," + base64Image;
       }
   </script>
</body>
</html>
